fn describe(m: Maybe<Tri>, cs: List<Codepoint>) -> Int {
    a = case m {
        Some(.Equilateral or .Isosceles) -> 1
        None or Some(_) -> 0
    }
    b = case cs {
        [')' or ']' as closer, ..rest] -> 1
        _ -> 0
    }
    c = case m {
        Some(.A(n)) or
            Some(.B(n)) as whole when n > 0 or n < -1 -> n
        (Some(.C(n)) as x) or (Some(.D(n)) as x) -> n
        _ -> 0
    }
    a + b + c
}

fn pick(.L(n) or .R(n): Two) -> Int {
    .L(m) or .R(m) = Two.L(n)
    Ok(v) = parse() else {
        Err(.Empty) or Err(.Blank) -> return 0
        Err(_) -> return 1
    }
    if 1 or 2 = v { m } else { v }
}
