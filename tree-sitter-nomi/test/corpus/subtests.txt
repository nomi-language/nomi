test "leap" {
  subtests cases, #(description, year), description {
    assert year > 0
  }
  subtests cases, #(a, b), "${a} + ${b}" {
    assert a < b
  }
  subtests rows, Ok(n) or Err(n), label(n) {
    assert n > 0
  }
  subtests cases, #(
    _description,
    year
  ), if year > 0 { "after" } else { "before" } {
    assert year != 0
  }
}
