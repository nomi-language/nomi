interface Scaled {
  fn scale(self(n) as t, by: Int) -> self where self: Distinct<Int> {
    self(m) = t
    case t {
      self(0) -> self(m)
      _ -> n |> self()
    }
  }

  fn all(_t: self, ns: List<Int>) -> Int where self: Distinct<Int> {
    ns |> Iter.map(self) |> Iter.count()
  }
}
