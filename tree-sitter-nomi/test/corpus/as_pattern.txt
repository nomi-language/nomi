An as pattern on the left side of a binding and nested in a pattern spelled as an expression

fn bindings(q: P, pair: ((Int, Int), Int), xs: List<Int>, m: Map<String, Int>): Int {
    {x: _c, y} as p = q
    {y, x: _c} as p = q
    {x: (a, b) as inner, y} as p = q
    ((a, b) as inner, c) as t = pair
    [x, ..rest] as all = xs else { return 0 }
    Some((a, b) as inner) as n = opt else { return 0 }
    {"a" => (a, b) as v} as all = m else { return 0 }
    0
}

fn nested_destructures(q: P, pair: ((Int, Int), Int)): Int {
    ((a, b), c) = pair
    ((a, b) as inner, c) = pair
    {x: (a, b), y} = q
    {x: Some(a), y} = q else { return 0 }
    {"a" => (a, b)} = m else { return 0 }
    0
}

fn arms(q: P, v: Option<(Int, Int)>): Int {
    case q {
        {x: (a, b) as xs, y} as p -> y
    }

    v |> case {
            Some((a, b) as p) -> a
            _ -> 0
        }
}

fn lambdas(xs: List<P>): Iter<Int> {
    xs |> Iter.map(|{x: (a, b) as t, y}| y)
}

test "pattern assert" {
    assert {x: _c, y} as p = q
    assert Some((a, b) as p) = v
}
