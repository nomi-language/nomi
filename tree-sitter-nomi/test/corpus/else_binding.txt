==========
A binding else takes a block
==========

fn main() {
  Some(email) = user.email else {
    return Err("none")
  }
  Some(e) = m else { "none" }
  Ok((w, h)) = parse(text) else { (80, 24) }
  [first, ..rest] = xs else { return 0 }
  .Valid(r) = c else { {addr: "none", score: 0} }
  {"a" => a} = m else { return 0 }
}

==========
A binding else takes case arms, and an if keeps its own else
==========

fn main() {
  Ok(user) = load(id) else {
    Err(.NotFound) -> return Err("no user")
    Err(e) when e == "x" -> 8080
    .Invalid{reason} -> continue
    _ -> break
  }
  label = if ok { "a" } else { "b" }
}
