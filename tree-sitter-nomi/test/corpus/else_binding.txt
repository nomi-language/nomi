fn main()
  Ok(user) = load(id) else
    Err(.NotFound) -> return Err("no user")
    Err(e) when e == "x" -> 8080
    .Invalid{reason:} -> continue
    _ -> break
  end
  label = if ok then "a" else "b"
end

================================================================================
Map, struct, list and or patterns in a binding with else
================================================================================

fn f(m: Map<String, Int>, p: Point, xs: List<Int>) -> Int
    {"port" => port} = m else
        {"fallback" => fallback} -> return fallback
        _ -> return 80
    end
    {x: 0, y:} = p else
        return 0
    end
    {done: True, n:} = p else
        return 0
    end
    [first, ..rest] = xs else
        return 0
    end
    {"a" => v} or {"b" => v} = m else
        return 0
    end
    port + y + n + first + v
end

struct Server

host fn now() -> Int

fn address(host: String) -> Server
    host = String.trim(host)
    {"host" => h} = config() else
        return Server{host:}
    end
    host
    Server{host: "${host}:${h}"}
end
